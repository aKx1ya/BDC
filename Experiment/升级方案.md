# BDC2026 Baseline 一步到位升级方案

> 基于三支获奖队伍经验 + 当前 baseline 代码全面审计，设计可落地的完整升级路线。
> 目标：将当前仅 StockTransformer 单模型的 baseline 升级为具备竞争力的多模型融合排序选股系统。

---

## 升级总览

```
当前 baseline                         升级后
───────────────                      ───────────────
单一 StockTransformer         →      XGBRanker + LGB Ranker + Transformer 三模型融合
绝对收益标签                  →      超额收益标签 + 排序标签（双标签可选）
单窗口 60 天                  →      多窗口 10/20/60 三窗口并行特征
一次性 80/20 验证             →      Walk-forward 滚动验证框架
直接 Top5 输出                →      Top30 召回 → 精排 Top5 两阶段
简单 softmax 权重             →      均值-方差优化 + 行业/相关性约束
无辅助任务                    →      方向预测 + 波动率刻画辅助任务
横截面特征默认关闭            →      横截面 + 行业 + 市场状态特征全面启用
```

## Phase 2：标签升级 —— 从绝对收益到超额收益排名

### 2.1 问题

当前 `train.py: _build_label_and_clean()` 中标签为：

```python
label = (open_t6 - open_t1) / open_t1  # 绝对收益
```

获奖队伍「O_O」和「7355608」一致指出：**绝对收益受大盘情绪干扰严重，模型极不稳定**。

### 2.2 方案

新增 `code/src/label.py`，提供多种标签构造策略：

```python
# code/src/label.py （新建文件）

import pandas as pd
import numpy as np

def build_absolute_return_label(df, buy_offset=1, sell_offset=6):
    """原始绝对收益标签（baseline 默认）"""
    df = df.copy()
    df['open_t1'] = df.groupby('股票代码')['开盘'].shift(-buy_offset)
    df['open_t6'] = df.groupby('股票代码')['开盘'].shift(-sell_offset)
    df['label'] = (df['open_t6'] - df['open_t1']) / (df['open_t1'] + 1e-12)
    return df.drop(columns=['open_t1', 'open_t6'])


def build_excess_return_label(df, buy_offset=1, sell_offset=6):
    """
    超额收益标签（推荐）
    label = 个股5日收益 - 沪深300等权平均收益
    获奖队伍 O_O 的核心经验：这比绝对收益稳定得多
    """
    df = df.copy()
    df['open_t1'] = df.groupby('股票代码')['开盘'].shift(-buy_offset)
    df['open_t6'] = df.groupby('股票代码')['开盘'].shift(-sell_offset)
    df['stock_return'] = (df['open_t6'] - df['open_t1']) / (df['open_t1'] + 1e-12)

    # 计算同一天所有股票的等权平均收益作为基准
    daily_mean = df.groupby('日期')['stock_return'].transform('mean')
    df['label'] = df['stock_return'] - daily_mean

    return df.drop(columns=['open_t1', 'open_t6', 'stock_return'])


def build_rank_label(df, buy_offset=1, sell_offset=6):
    """
    排序标签：同一天内收益率排名（0~1，越高越好）
    适合排序模型直接用 pct_rank 作为目标
    """
    df = df.copy()
    df['open_t1'] = df.groupby('股票代码')['开盘'].shift(-buy_offset)
    df['open_t6'] = df.groupby('股票代码')['开盘'].shift(-sell_offset)
    df['stock_return'] = (df['open_t6'] - df['open_t1']) / (df['open_t1'] + 1e-12)

    # 同一天内按收益率排名，转为 0~1 之间的分数
    df['label'] = df.groupby('日期')['stock_return'].rank(pct=True)

    return df.drop(columns=['open_t1', 'open_t6', 'stock_return'])


def build_direction_label(df, buy_offset=1, sell_offset=6):
    """
    方向标签（辅助任务用）：未来5日是否上涨（0/1）
    """
    df = df.copy()
    df['open_t1'] = df.groupby('股票代码')['开盘'].shift(-buy_offset)
    df['open_t6'] = df.groupby('股票代码')['开盘'].shift(-sell_offset)
    df['stock_return'] = (df['open_t6'] - df['open_t1']) / (df['open_t1'] + 1e-12)
    df['label'] = (df['stock_return'] > 0).astype(int)
    return df.drop(columns=['open_t1', 'open_t6', 'stock_return'])
```

## Phase 3：特征升级 —— 多尺度 + 横截面 + 行业 + 市场状态

### 3.1 问题

- 横截面特征默认关闭（`enable_extra_factors=0`）
- 无行业/板块特征
- 无市场宏观状态特征
- 所有特征仅在单一 60 日窗口内计算

### 3.2 方案

新建 `code/src/features.py`，实现完整的特征工程体系：

```python
# code/src/features.py （新建文件）

import pandas as pd
import numpy as np
from utils import (
    engineer_features_158plus39,
    add_extra_factor_features,
    add_cross_sectional_features,
)

# ============================================================
# 1. 行业特征
# ============================================================

# 申万一级行业映射表（内嵌，也可从外部文件加载）
SW_INDUSTRY_MAP = {
    '000001': '银行', '000002': '房地产',  # ... 需补全 300 只成分股
    # 实际使用时建议从文件加载
}

def add_industry_features(df, industry_map=None):
    """
    添加行业分类特征：
    - industry_code: 行业编码（用于 embedding 或 one-hot）
    - sector_return_3d/5d: 行业近3/5日等权平均收益
    - sector_volume_change: 行业成交额变化
    - stock_vs_sector_excess: 个股相对行业超额收益
    - sector_rank: 行业动能排名
    """
    if industry_map is None:
        industry_map = SW_INDUSTRY_MAP

    df = df.copy()
    df['industry'] = df['股票代码'].map(industry_map).fillna('未知')

    # 行业日内等权平均收益
    daily_sector_return = df.groupby(['日期', 'industry'])['涨跌幅'].transform('mean')
    df['sector_daily_return'] = daily_sector_return

    # 行业近3日、5日累计收益
    df['sector_return_3d'] = df.groupby('industry')['sector_daily_return'].transform(
        lambda x: x.rolling(3).mean()
    )
    df['sector_return_5d'] = df.groupby('industry')['sector_daily_return'].transform(
        lambda x: x.rolling(5).mean()
    )

    # 个股相对行业超额收益
    df['excess_vs_sector'] = df['涨跌幅'] - df['sector_daily_return']

    # 行业成交额变化
    daily_sector_amount = df.groupby(['日期', 'industry'])['成交额'].transform('sum')
    df['sector_amount'] = daily_sector_amount
    df['sector_amount_change_5d'] = df.groupby('industry')['sector_amount'].transform(
        lambda x: x.pct_change(5)
    )

    # 行业动能排名（同一天内各行业收益率排名）
    df['sector_momentum_rank'] = df.groupby('日期')['sector_return_5d'].rank(pct=True)

    # 是否为强势行业（前30%）
    df['is_strong_sector'] = (df['sector_momentum_rank'] > 0.7).astype(int)

    # 行业 one-hot 编码
    industry_dummies = pd.get_dummies(df['industry'], prefix='ind')
    df = pd.concat([df, industry_dummies], axis=1)

    return df


# ============================================================
# 2. 市场状态特征
# ============================================================

def add_market_features(df, index_df=None):
    """
    添加沪深300市场状态特征：
    - 如果 index_df 为 None，则从同一天所有股票的等权平均近似代替
    - market_return_1d/5d/20d
    - market_above_ma20
    - market_volatility_20d
    - advance_decline_ratio
    - market_volume_change
    """
    df = df.copy()

    if index_df is not None:
        # 使用真实沪深300指数数据
        market_return = index_df.set_index('日期')['涨跌幅']
    else:
        # 用等权平均近似
        market_return = df.groupby('日期')['涨跌幅'].mean()

    market_return.name = 'market_return'
    df = df.join(market_return, on='日期')

    # 市场近5日、20日累计收益
    df['market_return_5d'] = df.groupby('股票代码')['market_return'].transform(
        lambda x: x.rolling(5, min_periods=1).mean()
    )
    df['market_return_20d'] = df.groupby('股票代码')['market_return'].transform(
        lambda x: x.rolling(20, min_periods=1).mean()
    )

    # 市场波动率
    df['market_volatility_20d'] = df.groupby('股票代码')['market_return'].transform(
        lambda x: x.rolling(20, min_periods=1).std()
    )

    # 市场是否位于20日均线上方（用累积收益近似）
    df['market_above_ma20'] = (df['market_return_20d'] > 0).astype(int)

    # 涨跌比：同一天上涨股票数 / 下跌股票数
    daily_up = df.groupby('日期')['涨跌幅'].apply(lambda x: (x > 0).sum())
    daily_down = df.groupby('日期')['涨跌幅'].apply(lambda x: (x < 0).sum())
    daily_adr = (daily_up / (daily_down + 1)).rename('advance_decline_ratio')
    df = df.join(daily_adr, on='日期')

    # 市场成交额变化
    daily_total_amount = df.groupby('日期')['成交额'].sum().rename('market_total_amount')
    df = df.join(daily_total_amount, on='日期')
    df['market_amount_change_5d'] = df.groupby('股票代码')['market_total_amount'].transform(
        lambda x: x.pct_change(5)
    )

    return df


# ============================================================
# 3. 特征工程主入口
# ============================================================

def engineer_all_features(
    df,
    feature_mode='full',          # 'baseline' | 'full'
    enable_cross_sectional=True,  # 横截面特征
    enable_industry=True,         # 行业特征
    enable_market=True,           # 市场状态
    industry_map=None,
    index_df=None,
):
    """
    统一特征工程入口，替代原来分散的特征构造逻辑。
    """
    # Step 1: 基础量价特征（158+39 = 197维）
    df = engineer_features_158plus39(df)

    # Step 2: 横截面特征（排名/Z-score）
    if enable_cross_sectional:
        df = add_cross_sectional_features(df)

    # Step 3: 额外时序因子（VWAP偏离、log换手率等）
    if feature_mode == 'full':
        df = add_extra_factor_features(df)

    # Step 4: 行业特征
    if enable_industry:
        df = add_industry_features(df, industry_map)

    # Step 5: 市场状态特征
    if enable_market:
        df = add_market_features(df, index_df)

    # Step 6: 统一清理
    df.replace([np.inf, -np.inf], np.nan, inplace=True)
    df.fillna(0, inplace=True)

    return df
```

## Phase 4：模型升级 —— XGBRanker + LightGBM Ranker + Transformer 三模型融合

### 4.1 问题

当前只有一个 `StockTransformer`，获奖队伍一致使用树模型排序器（XGBRanker/LightGBM Ranker）作为核心或重要组件。

### 4.2 方案

新建 `code/src/models.py`，实现多模型训练与融合框架：

```python
# code/src/models.py （新建文件）

import numpy as np
import pandas as pd
import xgboost as xgb
import lightgbm as lgb
from sklearn.model_selection import TimeSeriesSplit


# ============================================================
# 1. XGBoost Ranker 包装器
# ============================================================

class XGBRankerWrapper:
    """
    XGBoost 排序模型包装器。

    XGBoost 的 rank:pairwise / rank:ndcg 目标天然适合排序任务。
    获奖队伍 O_O 的核心工具。
    """

    def __init__(self, params=None):
        default_params = {
            'objective': 'rank:ndcg',        # NDCG 排序目标（比 pairwise 更贴切 Top-K）
            'learning_rate': 0.05,
            'max_depth': 6,
            'subsample': 0.8,
            'colsample_bytree': 0.8,
            'min_child_weight': 10,
            'gamma': 0.1,
            'reg_alpha': 0.1,
            'reg_lambda': 1.0,
            'n_estimators': 500,
            'early_stopping_rounds': 30,
            'eval_metric': ['ndcg@5', 'ndcg@10'],
            'random_state': 42,
            'n_jobs': -1,
            'tree_method': 'hist',
            'device': 'cuda' if _cuda_available() else 'cpu',
        }
        self.params = {**default_params, **(params or {})}

    def fit(self, X_train, y_train, group_train, X_val, y_val, group_val):
        """训练 XGBRanker。

        参数:
        - X_train: (n_samples, n_features) 特征矩阵（已切分为表格形式，每行一只股票一个日期）
        - y_train: (n_samples,) 标签（超额收益或排序分）
        - group_train: list[int] 每个日期有多少只股票，如 [298, 295, 300, ...]
        - X_val, y_val, group_val: 同上，验证集
        """
        dtrain = xgb.DMatrix(X_train, label=y_train)
        dtrain.set_group(group_train)

        dval = xgb.DMatrix(X_val, label=y_val)
        dval.set_group(group_val)

        self.model = xgb.train(
            self.params,
            dtrain,
            num_boost_round=self.params['n_estimators'],
            evals=[(dtrain, 'train'), (dval, 'val')],
            early_stopping_rounds=self.params['early_stopping_rounds'],
            verbose_eval=50,
        )
        self.best_ntree = self.model.best_ntree_limit
        return self

    def predict(self, X):
        dtest = xgb.DMatrix(X)
        return self.model.predict(dtest, ntree_limit=self.best_ntree)

    def get_feature_importance(self):
        importance = self.model.get_score(importance_type='gain')
        return pd.Series(importance).sort_values(ascending=False)


# ============================================================
# 2. LightGBM Ranker 包装器
# ============================================================

class LGBRankerWrapper:
    """
    LightGBM 排序模型包装器。

    使用 lambdarank 目标，同样天然适合排序。
    与 XGBRanker 互补——两者树结构不同，融合效果好。
    """

    def __init__(self, params=None):
        default_params = {
            'objective': 'lambdarank',
            'metric': 'ndcg',
            'ndcg_eval_at': [5, 10, 30],
            'boosting_type': 'gbdt',
            'learning_rate': 0.05,
            'num_leaves': 63,
            'max_depth': 7,
            'subsample': 0.8,
            'colsample_bytree': 0.8,
            'min_child_samples': 20,
            'reg_alpha': 0.1,
            'reg_lambda': 1.0,
            'n_estimators': 500,
            'early_stopping_rounds': 30,
            'random_state': 42,
            'n_jobs': -1,
            'verbosity': -1,
        }
        self.params = {**default_params, **(params or {})}

    def fit(self, X_train, y_train, group_train, X_val, y_val, group_val):
        """训练 LGBMRanker。

        group_train 是每个 query（日期）的样本数列表。
        """
        dtrain = lgb.Dataset(X_train, label=y_train, group=group_train)
        dval = lgb.Dataset(X_val, label=y_val, group=group_val, reference=dtrain)

        self.model = lgb.train(
            self.params,
            dtrain,
            valid_sets=[dtrain, dval],
            valid_names=['train', 'val'],
            callbacks=[lgb.early_stopping(self.params['early_stopping_rounds']),
                       lgb.log_evaluation(50)],
        )
        self.best_iteration = self.model.best_iteration
        return self

    def predict(self, X):
        return self.model.predict(X, num_iteration=self.best_iteration)

    def get_feature_importance(self):
        importance = self.model.feature_importance(importance_type='gain')
        feature_names = self.model.feature_name()
        return pd.Series(importance, index=feature_names).sort_values(ascending=False)


# ============================================================
# 3. 多模型融合器
# ============================================================

class ModelEnsemble:
    """
    多模型分数融合器。

    策略：
    1. 每个模型对同一天的所有股票输出分数
    2. 分数在日截面内转为排名（rank）以消除量纲差异
    3. 加权平均各模型排名
    4. 输出融合后的排序

    获奖队伍 7355608 的经验：分数融合 + 后处理非常关键。
    """

    def __init__(self, model_weights=None):
        """
        model_weights: dict, 如 {'xgboost': 0.4, 'lightgbm': 0.35, 'transformer': 0.25}
        """
        self.model_weights = model_weights or {
            'xgboost': 0.4,
            'lightgbm': 0.35,
            'transformer': 0.25,
        }
        self.models = {}

    def add_model(self, name, model):
        self.models[name] = model

    def predict_and_ensemble(self, X, dates, stock_ids):
        """
        对每个日期的横截面做预测并融合。

        参数:
        - X: (n_samples, n_features) 特征矩阵
        - dates: (n_samples,) 每条样本对应的日期
        - stock_ids: (n_samples,) 每条样本对应的股票代码

        返回:
        - result_df: DataFrame with columns [date, stock_id, ensemble_score, rank]
        """
        all_scores = {}

        # Step 1: 每个模型分别预测
        for name, model in self.models.items():
            raw_scores = model.predict(X)
            all_scores[name] = raw_scores

        # Step 2: 日截面内排名归一化
        df = pd.DataFrame({
            'date': dates,
            'stock_id': stock_ids,
        })
        for name, scores in all_scores.items():
            df[f'{name}_score'] = scores

        # 每天内部排名（消除模型量纲差异）
        for name in self.models:
            df[f'{name}_rank'] = df.groupby('date')[f'{name}_score'].rank(pct=True)

        # Step 3: 加权融合
        df['ensemble_score'] = 0.0
        for name, weight in self.model_weights.items():
            if name in self.models:
                df['ensemble_score'] += weight * df[f'{name}_rank']

        # Step 4: 融合后排名
        df['rank'] = df.groupby('date')['ensemble_score'].rank(ascending=False)

        return df.sort_values(['date', 'rank'])

def prepare_tabular_data(sequences, targets, stock_ids_by_day):
    """
    将 3D 序列数据 (n_days, n_stocks, seq_len, n_features)
    展平为 2D 表格数据 (n_samples, n_features * seq_len)，
    供 XGBoost/LightGBM 使用。

    同时返回每天的样本数（group），用于排序模型的 query group。

    更好的做法：用序列最后 K 天的统计量（均值、std、趋势等）
    作为特征，而不是直接把整个序列展平（展平会导致特征维度过高）。
    """
    # 推荐：对每个序列取最后 K 天的统计量
    # 这里简化展示
    samples = []
    labels = []
    groups = []
    all_dates = []
    all_stock_ids = []

    for day_idx, (day_seq, day_target, day_stocks) in enumerate(
        zip(sequences, targets, stock_ids_by_day)
    ):
        n_stocks = len(day_stocks)
        # day_seq: (n_stocks, seq_len, n_features)
        # 展平: (n_stocks, seq_len * n_features)
        flat_features = day_seq.reshape(n_stocks, -1)
        samples.append(flat_features)
        labels.append(day_target)
        groups.append(n_stocks)
        all_dates.extend([f'day_{day_idx}'] * n_stocks)
        all_stock_ids.extend(day_stocks)

    X = np.vstack(samples)
    y = np.concatenate(labels)
    return X, y, groups, all_dates, all_stock_ids
```

## Phase 5：Walk-forward 验证框架

### 5.1 问题

当前只有一个静态的 80/20 时间切分。获奖队伍一致强调跨周期稳定性的重要性。

### 5.2 方案

新建 `code/src/validation.py`：

```python
# code/src/validation.py （新建文件）

import pandas as pd
import numpy as np
from datetime import timedelta


class WalkForwardValidator:
    """
    Walk-forward 滚动验证框架。

    每一步：
    1. 用过去 train_window 个交易日的数据训练
    2. 留 purge_gap 个交易日作为净化带
    3. 预测接下来 test_window 个交易日
    4. 记录该步的评分
    5. 整体窗口向前滚动 step_size 个交易日

    这直接对应方案.md 第 4 步的设计。
    """

    def __init__(
        self,
        train_window=252,       # 训练窗口：约1年交易日
        test_window=5,          # 测试窗口：5个交易日（比赛口径）
        purge_gap=5,            # 净化带：防止标签泄漏
        step_size=5,            # 滚动步长：每次前移5个交易日
        min_train_size=120,     # 最小训练样本数
        n_splits=None,          # 最多做几轮（None=能做多少做多少）
    ):
        self.train_window = train_window
        self.test_window = test_window
        self.purge_gap = purge_gap
        self.step_size = step_size
        self.min_train_size = min_train_size
        self.n_splits = n_splits

    def split(self, dates):
        """
        生成 Walk-forward 切分的 (train_dates, test_dates) 对。

        参数:
        - dates: 已排序的去重日期列表

        返回:
        - list of (train_start, train_end, test_start, test_end)
        """
        dates = sorted(pd.to_datetime(dates).unique())
        splits = []
        i = 0

        while True:
            # 测试集结束位置
            test_end_idx = len(dates) - i * self.step_size - 1
            test_start_idx = test_end_idx - self.test_window + 1

            if test_start_idx < self.min_train_size + self.train_window:
                break

            # 训练集结束位置（留 purge gap）
            train_end_idx = test_start_idx - self.purge_gap - 1
            train_start_idx = max(0, train_end_idx - self.train_window + 1)

            if train_end_idx - train_start_idx < self.min_train_size:
                break

            splits.append({
                'fold': len(splits),
                'train_dates': dates[train_start_idx:train_end_idx + 1],
                'test_dates': dates[test_start_idx:test_end_idx + 1],
                'train_start': dates[train_start_idx],
                'train_end': dates[train_end_idx],
                'test_start': dates[test_start_idx],
                'test_end': dates[test_end_idx],
            })

            i += 1
            if self.n_splits and len(splits) >= self.n_splits:
                break

        return splits

    def run(self, train_fn, predict_fn, score_fn, data_df, verbose=True):
        """
        执行完整的 Walk-forward 验证。

        参数:
        - train_fn: function(train_df, fold_info) -> model
        - predict_fn: function(model, test_df) -> result_df  (stock_id, weight)
        - score_fn: function(result_df, test_df) -> float   (Final Score)
        - data_df: 完整数据

        返回:
        - results: list of dict, 每轮的结果
        """
        dates = sorted(data_df['日期'].unique())
        splits = self.split(dates)
        results = []

        for fold_info in splits:
            train_mask = data_df['日期'].isin(fold_info['train_dates'])
            test_mask = data_df['日期'].isin(fold_info['test_dates'])

            train_df = data_df[train_mask].copy()
            test_df = data_df[test_mask].copy()

            # 训练
            model = train_fn(train_df, fold_info)

            # 预测
            result_df = predict_fn(model, test_df)

            # 评分
            score = score_fn(result_df, test_df)

            results.append({
                'fold': fold_info['fold'],
                'train_period': f"{fold_info['train_start'].date()} ~ {fold_info['train_end'].date()}",
                'test_period': f"{fold_info['test_start'].date()} ~ {fold_info['test_end'].date()}",
                'score': score,
            })

            if verbose:
                print(f"Fold {fold_info['fold']}: "
                      f"train={fold_info['train_start'].date()}~{fold_info['train_end'].date()}, "
                      f"test={fold_info['test_start'].date()}~{fold_info['test_end'].date()}, "
                      f"score={score:.6f}")

        # 汇总
        scores = [r['score'] for r in results]
        summary = {
            'n_folds': len(results),
            'mean_score': np.mean(scores),
            'std_score': np.std(scores),
            'min_score': np.min(scores),
            'max_score': np.max(scores),
            'sharpe_like': np.mean(scores) / (np.std(scores) + 1e-12),
            'folds': results,
        }

        if verbose:
            print(f"\n=== Walk-forward 汇总 ===")
            print(f"轮次: {summary['n_folds']}")
            print(f"平均得分: {summary['mean_score']:.6f} ± {summary['std_score']:.6f}")
            print(f"得分范围: [{summary['min_score']:.6f}, {summary['max_score']:.6f}]")
            print(f"稳定性(Sharpe-like): {summary['sharpe_like']:.2f}")

        return summary
```

### 6.1 问题

当前 `predict.py` 直接从全部 300 只股票选 Top5，缺少中间候选池和精排环节。

### 6.2 方案

新建 `code/src/postprocess.py`：

```python
# code/src/postprocess.py （新建文件）

import pandas as pd
import numpy as np
from scipy.optimize import minimize


# ============================================================
# 1. 硬门控过滤（Hard Gates）
# ============================================================

def apply_hard_gates(candidates_df, config=None):
    """
    对候选池进行硬门控过滤。

    过滤条件（可配置）：
    - 近3日平均成交额 < 门槛 → 剔除
    - 近20日最大回撤 > 门槛 → 剔除
    - 停牌 / 异常波动 → 剔除
    """
    df = candidates_df.copy()
    initial_count = len(df)

    # 流动性门槛：近3日平均成交额 >= 1亿（可配置）
    if 'avg_amount_3d' in df.columns:
        amount_threshold = 1e8  # 1亿
        df = df[df['avg_amount_3d'] >= amount_threshold]

    # 回撤门槛：近20日最大回撤 < 15%（可配置）
    if 'max_drawdown_20d' in df.columns:
        dd_threshold = 0.15
        df = df[df['max_drawdown_20d'] < dd_threshold]

    # 涨跌幅异常过滤
    if '涨跌幅' in df.columns:
        df = df[df['涨跌幅'].abs() < 9.5]  # 排除涨跌停（近似）

    filtered_count = initial_count - len(df)
    if filtered_count > 0:
        print(f"[Hard Gate] 过滤掉 {filtered_count}/{initial_count} 只股票")

    return df


# ============================================================
# 2. 行业集中度约束
# ============================================================

def apply_sector_constraint(ranked_df, max_per_sector=2, industry_col='industry'):
    """
    行业分散度约束：同一行业最多 max_per_sector 只股票。

    按排名从高到低遍历，遇到行业已满则跳过。
    """
    selected = []
    sector_counts = {}

    for _, row in ranked_df.iterrows():
        sector = row.get(industry_col, '未知')
        if sector_counts.get(sector, 0) < max_per_sector:
            selected.append(row)
            sector_counts[sector] = sector_counts.get(sector, 0) + 1

    return pd.DataFrame(selected)


# ============================================================
# 3. 相关性约束
# ============================================================

def apply_correlation_constraint(ranked_df, returns_df, max_corr=0.3, top_k=5):
    """
    相关性约束：与已入选股票的历史收益相关性 > max_corr 则跳过。

    参数:
    - ranked_df: 已按分数排序的候选股票
    - returns_df: 各股票的历史日收益序列（行=日期，列=股票代码）
    - max_corr: 最大允许的相关系数
    - top_k: 最终选几只
    """
    selected = []
    selected_returns = []

    for _, row in ranked_df.iterrows():
        stock_id = row['stock_id']
        if stock_id not in returns_df.columns:
            continue

        stock_ret = returns_df[stock_id].dropna().values

        # 检查与已选股票的相关性
        too_correlated = False
        for sel_ret in selected_returns:
            min_len = min(len(stock_ret), len(sel_ret))
            if min_len < 10:
                continue
            corr = np.corrcoef(stock_ret[-min_len:], sel_ret[-min_len:])[0, 1]
            if abs(corr) > max_corr:
                too_correlated = True
                break

        if not too_correlated:
            selected.append(row)
            selected_returns.append(stock_ret)

        if len(selected) >= top_k:
            break

    return pd.DataFrame(selected)


# ============================================================
# 4. 均值-方差权重优化
# ============================================================

def optimize_weights_mean_variance(
    expected_returns,
    cov_matrix,
    max_weight=0.45,
    min_weight=0.0,
    risk_aversion=1.0,
):
    """
    均值-方差优化分配权重。

    目标: max(w·μ - λ·w·Σ·w)
    约束: Σw ≤ 1, 0 ≤ w_i ≤ max_weight

    参数:
    - expected_returns: (n,) 预期收益向量（来自模型分数映射）
    - cov_matrix: (n, n) 协方差矩阵
    - max_weight: 单只股票最大权重
    - min_weight: 单只股票最小权重
    - risk_aversion: 风险厌恶系数 λ

    返回:
    - weights: (n,) 最优权重
    """
    n = len(expected_returns)

    def objective(w):
        return -(np.dot(w, expected_returns) - risk_aversion * np.dot(w, np.dot(cov_matrix, w)))

    constraints = [
        {'type': 'ineq', 'fun': lambda w: 1.0 - np.sum(w)},   # Σw ≤ 1
    ]
    bounds = [(min_weight, max_weight) for _ in range(n)]
    x0 = np.ones(n) / n  # 初始等权

    result = minimize(
        objective, x0, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-12},
    )

    weights = result.x
    weights = np.clip(weights, 0, max_weight)
    weights = weights / weights.sum() * min(1.0, weights.sum())
    return weights


# ============================================================
# 5. 精排主流程
# ============================================================

def fine_ranking(
    candidate_df,        # Top30 候选池 (stock_id, ensemble_score, ...)
    price_df,            # 价格数据（用于计算回撤等）
    returns_df,          # 日收益矩阵
    sector_map,          # 行业映射
    config=None,
):
    """
    精排主流程：Top30 → Top5

    步骤：
    1. Hard Gates 硬门控过滤
    2. 多维打分（ML分数 + 行业动能 + 价格行为 + 估值）
    3. 行业集中度约束
    4. 相关性约束
    5. 权重优化
    """
    df = candidate_df.copy()

    # Step 1: Hard Gates
    df = apply_hard_gates(df, config)

    # Step 2: 多维打分
    # ML 分数来自模型融合（已在 candidate_df 中）
    # 此处可叠加行业动能、价格行为等分数
    if 'sector_momentum_rank' in df.columns:
        df['refine_score'] = (
            0.40 * df['ensemble_rank_pct'] +
            0.30 * df['sector_momentum_rank'] +
            0.15 * df.get('price_action_score', 0) +
            0.15 * df.get('style_score', 0)
        )
    else:
        df['refine_score'] = df['ensemble_rank_pct']

    df = df.sort_values('refine_score', ascending=False)

    # Step 3: 行业约束
    if 'industry' in df.columns:
        df = apply_sector_constraint(df, max_per_sector=2)

    # Step 4: 相关性约束
    df = apply_correlation_constraint(df, returns_df, max_corr=0.3, top_k=5)

    # Step 5: 权重优化
    top5 = df.head(5)
    if len(top5) >= 2 and returns_df is not None:
        # 用模型分数映射为预期收益
        expected_ret = top5['refine_score'].values
        expected_ret = (expected_ret - expected_ret.min()) / (expected_ret.max() - expected_ret.min() + 1e-8)

        # 计算协方差矩阵
        ret_data = returns_df[top5['stock_id'].values].dropna()
        if len(ret_data) > 20:
            cov = ret_data.cov().values
            weights = optimize_weights_mean_variance(expected_ret, cov)
        else:
            weights = np.ones(len(top5)) / len(top5)
    else:
        weights = np.ones(len(top5)) / len(top5)

    # 构造最终输出
    result = pd.DataFrame({
        'stock_id': top5['stock_id'].values,
        'weight': weights,
    })

    return result


# ============================================================
# 6. 辅助：模型分数 → 预期收益映射
# ============================================================

def scores_to_expected_returns(scores, method='rank_pct'):
    """将模型排序分数映射为预期收益估计"""
    if method == 'rank_pct':
        # 排名百分位作为预期收益代理
        from scipy.stats import rankdata
        ranks = rankdata(scores)
        return ranks / len(ranks)
    elif method == 'standardize':
        return (scores - np.mean(scores)) / (np.std(scores) + 1e-8)
    else:
        return scores
```

class StockTransformer(nn.Module):
    def __init__(self, input_dim, config, num_stocks):

            )

        if config.get('use_aux_volatility', False):

                nn.Linear(config['d_model'] // 2, config['d_model'] // 4),
                nn.ReLU(),
                nn.Dropout(config['dropout'] * 0.3),
                nn.Linear(config['d_model'] // 4, 1),  # 输出波动率预测
            )

    def forward(self, src, return_aux=False):
        # ... 原有前向传播得到 ranking_features (batch*num_stocks, d_model//2) ...

        # 主任务：排序分数

        if not return_aux:
            return scores.view(batch_size, num_stocks)

        if hasattr(self, 'volatility_head'):

def compute_multitask_loss(outputs, aux_outputs, targets, aux_targets, criterion, config):
    """多任务联合损失"""
    # 主任务：排序损失
    rank_loss = criterion(outputs, targets)

    total_loss = rank_loss

    # 辅助任务1：方向预测
    if 'direction' in aux_outputs and 'direction' in aux_targets:
        direction_loss = F.binary_cross_entropy_with_logits(
            aux_outputs['direction'].squeeze(),

            aux_targets['volatility'].float()
        )
        volatility_weight = config.get('aux_volatility_weight', 0.05)
        total_loss = total_loss + volatility_weight * volatility_loss

    return total_loss
